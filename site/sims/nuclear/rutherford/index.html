  • բռնկում էկրանին
Քաշեք մանրադիտակը էկրանի շրջանագծով

բռնկումների թիվը 2°-անոց միջակայքերում (ուղղաձիգ առանցքը՝ լոգարիթմական) տեսական կոր՝ 1/sin⁴(θ/2) մանրադիտակի անկյունը

Ռեզերֆորդի բանաձևը. նույն մակերեսով էկրանի տեղամասին ընկնող α-մասնիկների թիվը

N(90°) / N(10°) ≈ 1 / 4300
N(180°) / N(10°) ≈ 1 / 17 000

Ինչ ցույց տվեց փորձը

1909–1911 թվականներին Ռեզերֆորդի աշխատակիցներ Հ. Գայգերը և Է. Մարսդենը ռադիոակտիվ աղբյուրից դուրս եկող α-մասնիկների նեղ փունջն ուղղում էին ոսկու շատ բարակ նրբաթիթեղի վրա։ Ցրված մասնիկներն ընկնում էին ցինկի սուլֆիդով պատված էկրանին և առաջացնում թույլ բռնկումներ, որոնք հաշվում էին մանրադիտակով՝ այն պտտելով նրբաթիթեղի շուրջը։ Մասնիկների գերակշիռ մասն անցնում էր նրբաթիթեղով գրեթե չշեղվելով, սակայն մոտավորապես ութ հազարից մեկը շեղվում էր 90°-ից ավելի, իսկ որոշները՝ ետ էին շպրտվում։

Թոմսոնի մոդելում դրական լիցքը «քսված» է ամբողջ ատոմով մեկ, և նրա էլեկտրական դաշտն այնքան թույլ է, որ α-մասնիկը յուրաքանչյուր ատոմում շեղվում է ընդամենը հարյուրերորդական աստիճաններով։ Հազարավոր ատոմներով անցնելիս այդ շեղումները գումարվում են պատահականորեն և տալիս մոտ 1°-ի կարգի ցրում. մեծ անկյուններով ցրված մասնիկներ գործնականում չպետք է լինեին։ Ռեզերֆորդն այդ առթիվ ասել է, որ դա գրեթե նույնքան անհավանական էր, որքան եթե 15 դյույմանոց արկով կրակեիք բարակ թղթի թերթիկի վրա, իսկ արկը հետ դառնար ու խփեր ձեզ։

Եզրակացությունը. ատոմի ամբողջ դրական լիցքը և զանգվածի գրեթե ամբողջ մասը կենտրոնացած են շատ փոքր՝ մոտ 10⁻¹⁴ մ չափով միջուկում, որը մոտ 10 000 անգամ փոքր է ատոմից, իսկ ատոմի մնացած մասը գրեթե դատարկ է։ Միայն միջուկին շատ մոտ անցնող հազվադեպ մասնիկներն են մեծ անկյուններով շեղվում, և նրանց թիվը նվազում է ճիշտ 1/sin⁴(θ/2) օրենքով։ Փորձեք. տեղադրեք մանրադիտակը 30°, 90° և 150° անկյուններում և համեմատեք բռնկումների հաճախությունը, ապա միացրեք Թոմսոնի մոդելը։

α-մասնիկի հետագիծ ցրման անկյուն θ rmin՝ մոտեցման նվազագույն հեռավորություն d₀ շառավղով շրջանագիծ

α-մասնիկը միջուկի դաշտում

Դրական լիցքավորված α-մասնիկը (լիցքը՝ 2e) վանվում է միջուկից (լիցքը՝ Ze) Կուլոնի ուժով, որը հակադարձ համեմատական է հեռավորության քառակուսուն։ Այդ ուժի ազդեցությամբ մասնիկը շարժվում է հիպերբոլով։ Ինչքան մոտ է անցնում մասնիկը միջուկին, այսինքն՝ ինչքան փոքր է նշանառության պարամետրը b (մասնիկի սկզբնական ուղղի հեռավորությունը միջուկից), այնքան մեծ է ցրման անկյունը՝ tg(θ/2) = d₀ / (2b)։ Ճակատային բախման դեպքում (b = 0) մասնիկը կանգ է առնում d₀ հեռավորության վրա, որտեղ նրա ամբողջ կինետիկ էներգիան վերածվում է կուլոնյան պոտենցիալ էներգիայի, և ետ է շպրտվում (θ = 180°)։

Ոսկու համար 5 ՄէՎ էներգիայի դեպքում d₀ ≈ 45 ֆմ, իսկ ատոմի չափը մոտ 100 000 ֆմ է։ Ուստի մեծ անկյուններով ցրվում են միայն այն մասնիկները, որոնք «նշան են բռնում» ատոմի կենտրոնի մի չնչին տիրույթ։ Թոմսոնի մոդելում դրական լիցքը բաշխված է ատոմի ամբողջ ծավալով, ուստի պատկերի մասշտաբը 1000 անգամ մեծ է (պիկոմետրեր), իսկ ցրման անկյունները չեն գերազանցում հարյուրերորդական աստիճանները։

Ռեզերֆորդի բանաձևը ճիշտ է, քանի դեռ α-մասնիկը չի դիպչում միջուկին։ Ընտրեք ալյումին և մեծացրեք էներգիան. երբ rmin-ը դառնում է միջուկի և α-մասնիկի շառավիղների գումարից փոքր, միանում են միջուկային ուժերը, և ցրումը շեղվում է Ռեզերֆորդի օրենքից։ Հենց այդ շեղումներից էլ առաջին անգամ գնահատվեցին միջուկների չափերը։
